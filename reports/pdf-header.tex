% LaTeX preamble for pandoc -> PDF compilation (CEU MSBA capstone)
% Implements CEU thesis formatting requirements per university guidance.
% Use with xelatex (required for system fonts like Calibri).

% --------------------------------------------------------------
% Single-page landscape orientation (used to give wide figures more room).
% On landscape pages, the page number is suppressed because pdflscape's
% default behaviour rotates the footer along with the content, placing
% the number on the visual left margin (where it reads vertically), which
% is worse than no number at all. The page is still counted in the total
% page sequence.
% --------------------------------------------------------------
\usepackage{pdflscape}
\AtBeginEnvironment{landscape}{\thispagestyle{empty}}

% --------------------------------------------------------------
% Text rendering: microtype improves justification and reduces awkward
% hyphenations via character protrusion + font expansion.
% Explicit no-hyphenation rule for words that have hyphenated awkwardly
% (classifier and its forms have broken across lines as "clas-sifier").
% --------------------------------------------------------------
\usepackage{microtype}
\hyphenation{classifier classifiers}

% --------------------------------------------------------------
% Table formatting
% --------------------------------------------------------------
\usepackage[table]{xcolor}
\usepackage{array}
\usepackage{booktabs}
\usepackage{longtable}
\rowcolors{2}{gray!12}{white}
\renewcommand{\arraystretch}{1.25}


% --------------------------------------------------------------
% Spacing (CEU requires double-space in body, single in frontmatter,
% bibliography, footnotes, captions)
% Default: double-space body. Wrap frontmatter content in markdown using
% raw LaTeX \begin{singlespace}...\end{singlespace} blocks.
% --------------------------------------------------------------
\usepackage{setspace}
\doublespacing

% --------------------------------------------------------------
% Widow / orphan / headers-near-tables control
% --------------------------------------------------------------
\widowpenalty=10000
\clubpenalty=10000
\usepackage{needspace}
% Manually insert \needspace{6\baselineskip} before a table in markdown
% (via raw LaTeX) to keep the heading above with the table.

% --------------------------------------------------------------
% Captions (CEU: single-spaced). Table captions positioned BELOW the table.
% --------------------------------------------------------------
\usepackage[font=small,labelfont=bf,labelsep=colon,justification=centering,singlelinecheck=false]{caption}
\captionsetup{font={small,singlespacing}}
\captionsetup[table]{position=below}
\captionsetup[longtable]{position=below}

% --------------------------------------------------------------
% Combine List of Figures and List of Tables into a single section
% titled "List of Figures and Tables", with "Figures" and "Tables"
% subheadings inside.
% --------------------------------------------------------------
\renewcommand{\listfigurename}{List of Figures and Tables}
\let\oldlistoftables\listoftables
\renewcommand{\listoftables}{%
  \begingroup
  \renewcommand{\chapter}[2]{}% swallow the chapter heading
  \oldlistoftables
  \endgroup
}
\let\oldlistoffigures\listoffigures
\renewcommand{\listoffigures}{%
  \oldlistoffigures
  \vspace{0.8em}
  \noindent{\large\bfseries Tables}\par\nobreak\vspace{0.3em}
  \listoftables
}
% Prepend "Figures" subheading to the figure list section.
% Injection into the auxiliary .lof file via a hook; \nobreak prevents
% LaTeX from inserting extra blank lines around the subheading.
\AtBeginDocument{%
  \addtocontents{lof}{\protect\noindent {\large\bfseries Figures}\protect\par\protect\nobreak\protect\vspace{0.3em}}%
}

% --------------------------------------------------------------
% Footnotes (CEU: 10pt, single-spaced)
% --------------------------------------------------------------
\usepackage[hang,bottom,marginal,multiple,flushmargin]{footmisc}
\setlength{\footnotesep}{0.7em}
\renewcommand{\footnotesize}{\fontsize{10pt}{12pt}\selectfont}

% --------------------------------------------------------------
% Paragraphs: indented (CEU allows indent OR empty line; indent chosen).
% indentfirst forces the first paragraph after every heading to also
% be indented (LaTeX default skips it).
% --------------------------------------------------------------
\setlength{\parindent}{1.27cm}
\usepackage{indentfirst}

% --------------------------------------------------------------
% Chapter heading spacing: remove the report-class default top gap
% so chapter titles start near the top margin (still honoring the 2.5cm
% page margin).
% --------------------------------------------------------------
\usepackage{titlesec}
\titleformat{\chapter}[hang]
  {\normalfont\Huge\bfseries}{\thechapter.}{12pt}{}
\titlespacing*{\chapter}{0pt}{0pt}{20pt}
% Centered 18pt bold for unnumbered (front-matter) chapters, 12pt after-spacing
% (CEU spec is 24pt; reduced to 12pt to limit wasted vertical space on short pages).
\titleformat{name=\chapter,numberless}[hang]
  {\normalfont\fontsize{18}{22}\selectfont\bfseries\centering}{}{0pt}{}
\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{12pt}

% Per-chapter table and figure numbering: \counterwithin{table}{chapter}
% is loaded AFTER titlesec since titlesec's \chapter redefinition breaks
% the standard report.cls child-counter reset behaviour when loaded before.
%
% KNOWN ISSUE (intentionally accepted): tables and figures still render as
% "Table 0.X" and "Figure 0.X" rather than per-chapter (Table 5.1, Table 7.1,
% etc.). The \thechapter counter evaluates to 0 at caption-rendering time
% even though the chapter heading itself is correctly numbered, and multiple
% reset-pattern fixes (\@addtoreset, chngcntr, ordering before vs after
% titlesec) did not resolve it. Inline cross-references in the body prose
% use the "0.X" form to match the rendered captions, so cross-references
% resolve consistently. Fixing requires a deeper LaTeX-template change
% (likely replacing titlesec or restructuring \chapter handling) that was
% not in scope for the submission deadline.
\usepackage{chngcntr}
\counterwithin{table}{chapter}
\counterwithin{figure}{chapter}

% --------------------------------------------------------------
% Suppress Pandoc's default \maketitle so the custom CEU titlepage
% block in the markdown is the only one rendered.
% --------------------------------------------------------------
\renewcommand{\maketitle}{}

% --------------------------------------------------------------
% Hyperlinks + PDF metadata + bookmarks
% --------------------------------------------------------------
% PDF metadata (link colors are passed via Pandoc -V flags so the template
% omits its default hidelinks option).
\AtBeginDocument{%
  \hypersetup{%
    pdfsubject={CEU MSBA Capstone 2026},
    pdfkeywords={enterprise AI agents, prompt injection, LLM-as-judge, layered defense, LoRA fine-tuning},
    bookmarksnumbered=true,
    bookmarksopen=true
  }%
}

% --------------------------------------------------------------
% List of Figures and List of Tables: rendered after TOC
% (Pandoc YAML: lof: true, lot: true)
% --------------------------------------------------------------
% No additional preamble needed; controlled via Pandoc.

% --------------------------------------------------------------
% Bibliography (References list): single-spaced
% Applied via \setstretch{1} around the bibliography section
% --------------------------------------------------------------
\AtBeginEnvironment{thebibliography}{\setstretch{1}}
